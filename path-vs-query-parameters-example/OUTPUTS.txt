### 3. Arrays and repeated keys
$ curl -s "http://localhost:8480/orders?status=paid&status=shipped"
{"status":["paid","shipped"],"sort":"-created","limit":20,"offset":0,"data":[{"id":1004,"customer_id":8,"status":"paid","total":25.0,"created":"2026-10-04"},{"id":1002,"customer_id":8,"status":"shipped","total":64.0,"created":"2026-10-02"},{"id":1001,"customer_id":7,"status":"paid","total":49.0,"created":"2026-10-01"}]}

$ curl -s "http://localhost:8480/orders?status=paid,shipped"
{"detail":[{"type":"literal_error","loc":["query","status",0],"msg":"Input should be 'pending', 'paid', 'shipped' or 'cancelled'","input":"paid,shipped","ctx":{"expected":"'pending', 'paid', 'shipped' or 'cancelled'"}}]}

### 4. Encoding
$ curl -s "http://localhost:8480/products?q=C++"
{"q":"C  ","category":null,"data":[]}

$ curl -s "http://localhost:8480/products?q=C%2B%2B"
{"q":"C++","category":null,"data":[{"sku":"KB-101","name":"C++ Primer","category":"books","price_usd":49.0}]}

$ curl -s "http://localhost:8480/products?q=AT&T"
{"q":"AT","category":null,"data":[{"sku":"HW-202","name":"AT&T Modem","category":"hardware","price_usd":80.0}]}

$ curl -s "http://localhost:8480/products?q=AT%26T"
{"q":"AT&T","category":null,"data":[{"sku":"HW-202","name":"AT&T Modem","category":"hardware","price_usd":80.0}]}

$ curl -s -G http://localhost:8480/products --data-urlencode "q=AT&T"
{"q":"AT&T","category":null,"data":[{"sku":"HW-202","name":"AT&T Modem","category":"hardware","price_usd":80.0}]}

$ curl -s http://localhost:8480/products/KB-101
{"sku":"KB-101","name":"C++ Primer","category":"books","price_usd":49.0}

$ curl -s http://localhost:8480/products/ZZ-999

$ curl -s http://localhost:8480/products/AB%2F12

### 6. URL length limits
$ long_value=$(head -c 8150 /dev/zero | tr '\0' x)
$ curl -s -o /dev/null -w "%{http_code}\n" "http://localhost:8481/orders?note=$long_value"
200

$ long_value=$(head -c 8170 /dev/zero | tr '\0' x)
$ curl -s -o /dev/null -w "%{http_code}\n" "http://localhost:8481/orders?note=$long_value"
414

$ # the same 8170-character value sent to uvicorn without nginx
$ curl -s -o /dev/null -w "%{http_code}\n" "http://localhost:8480/orders?note=$long_value"
200

$ curl -s "http://localhost:8481/orders?note=$long_value" | head -4
<html>
<head><title>414 Request-URI Too Large</title></head>
<body>
<center><h1>414 Request-URI Too Large</h1></center>
